```dart
// In supabase_service.dart callFunction:
const _coldStartBackoffsMs = [1500, 4000];  // 2 retries, exponential-ish

Future<FunctionResponse> callFunction(...) async {
  FunctionResponse response;
  int attempt = 0;
  while (true) {
    try {
      response = await client.functions.invoke(name, headers: headers, body: body);
      return response;
    } on FunctionException catch (e) {
      final isColdStart = e.status == 502 || e.status == 503;
      if (!isColdStart || attempt >= _coldStartBackoffsMs.length) {
        rethrow;
      }
      final backoff = _coldStartBackoffsMs[attempt];
      unawaited(ErrorTelemetry.logEvent(
        'edge_function_cold_start_retry',
        message: 'fn=$name attempt=${attempt + 1} status=${e.status} backoff_ms=$backoff',
      ));
      await Future<void>.delayed(Duration(milliseconds: backoff));
      attempt++;
    }
  }
}
```

```dart
Map<String, dynamic> _getTodayWorkout() {
  final today = istDateStr(DateTime.now());
  final schedule = _workoutBox.get('schedule_$today') as Map?;
  final type = schedule?['type'] ?? 'rest';
  final status = schedule?['status'] ?? 'scheduled';

  // Read LOGGED exercises (Option A — locked with founder 2026-05-12)
  // SoT: WorkoutReceiptData.fromExerciseLogs style — exercise_log_index_<date> + exlog_*
  final logIndex = _workoutBox.get('exercise_log_index_$today') as List?;
  final logged = <Map<String, dynamic>>[];
  if (logIndex != null) {
    for (final key in logIndex) {
      final raw = _workoutBox.get(key);
      if (raw is! Map) continue;
      final log = Map<String, dynamic>.from(raw);
      final sets = (log['sets'] as List?) ?? const [];
      final reps = sets.fold<int>(0, (a, s) => a + ((s as Map)['reps'] as num? ?? 0).toInt());
      final topWeight = sets
          .map((s) => ((s as Map)['weight_kg'] as num? ?? 0).toDouble())
          .fold<double>(0, (a, b) => b > a ? b : a);
      logged.add({
        'name': log['exercise_name'],
        'sets': sets.length,
        'reps_total': reps,
        'top_set_weight_kg': topWeight,
        'logging_type': log['logging_type'],
        'is_pr': log['is_pr'] ?? false,
      });
    }
  }

  return {
    'type': type,
    'status': status,
    'exercises': logged,  // ← LOGGED, not planned
  };
}
```
